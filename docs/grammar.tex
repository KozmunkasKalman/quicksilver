$$
\begin{align}
  [\text{prog}] &\to [\text{stmt}]^* \\
  [\text{stmt}] &\to
    \begin{cases}
      \text{exit} \space [\text{expr}] \\
      \text{int} \space \text{ident} [expr] \\
    \end{cases} \\
  [\text{expr}] &\to
    \begin{cases}
      \text{term} \\
      \text{bin\_expr} \\
    \end{cases} \\
  [\text{bin\_expr}] &\to
    \begin{cases}
      \text{expr} \space \text{expr} \text{prec} = 1\\
      \text{expr} / \text{expr} \text{prec} = 1\\
      \text{expr} + \text{expr} \text{prec} = 0\\
      \text{expr} - \text{expr} \text{prec} = 0\\
    \end{cases} \\
  [\text{term}] &\to
    \begin{cases}
      \text{int\_lit} \\
      \text{ident} \\
    \end{cases}
\end{align}
$$
